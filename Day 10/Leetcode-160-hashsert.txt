public class Solution {
    public ListNode getIntersectionNode(ListNode headA, ListNode headB) {
        Set<ListNode> nodesInA = new HashSet<>();
        while (headA != null) {
            nodesInA.add(headA);
            headA = headA.next;
        }
        while (headB != null) {
            if (nodesInA.contains(headB)) return headB;
            headB = headB.next;
        }
        return null;
    }
}
